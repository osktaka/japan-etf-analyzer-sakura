```python
# backend/tests/unit/services/test_etf_service.py

import pytest
from src.services.etf_service import ETFService
from src.models.etf import ETF

class TestETFService:
    """ETFServiceの単体テスト"""

    def test_search_by_keyword(self, app, sample_etf):
        """キーワード検索のテスト"""
        with app.app_context():
            # Arrange
            ETF.create(**sample_etf)

            # Act
            result = ETFService.search({'q': 'TOPIX'})

            # Assert
            assert len(result['items']) == 1
            assert result['items'][0]['code'] == '1306'

    def test_search_by_category(self, app, sample_etf):
        """カテゴリ検索のテスト"""
        with app.app_context():
            # Arrange
            ETF.create(**sample_etf)

            # Act
            result = ETFService.search({'category_id': 1})

            # Assert
            assert len(result['items']) == 1

    def test_search_empty_result(self, app):
        """検索結果0件のテスト"""
        with app.app_context():
            # Act
            result = ETFService.search({'q': 'nonexistent'})

            # Assert
            assert len(result['items']) == 0
            assert result['meta']['total'] == 0

    def test_get_by_code_found(self, app, sample_etf):
        """銘柄コードで取得（存在する場合）"""
        with app.app_context():
            # Arrange
            ETF.create(**sample_etf)

            # Act
            result = ETFService.get_by_code('1306')

            # Assert
            assert result is not None
            assert result['code'] == '1306'

    def test_get_by_code_not_found(self, app):
        """銘柄コードで取得（存在しない場合）"""
        with app.app_context():
            # Act
            result = ETFService.get_by_code('9999')

            # Assert
            assert result is None
```

### 2.4 カスタムフックテスト例

```typescript
// frontend/src/hooks/useETFSearch.test.ts

import { renderHook, waitFor } from '@testing-library/react';
import { describe, it, expect, vi } from 'vitest';
import { useETFSearch } from './useETFSearch';
import { etfApi } from '../api/etf';

vi.mock('../api/etf');

describe('useETFSearch', () => {
  it('初期状態ではデータが空', () => {
    // Act
    const { result } = renderHook(() => useETFSearch());

    // Assert
    expect(result.current.data).toEqual([]);
    expect(result.current.isLoading).toBe(false);
  });

  it('検索実行時にAPIが呼ばれる', async () => {
    // Arrange
    const mockData = { data: [{ code: '1306' }], meta: { total: 1 } };
    vi.mocked(etfApi.search).mockResolvedValue(mockData);

    // Act
    const { result } = renderHook(() => useETFSearch());
    result.current.search({ q: 'TOPIX' });

    // Assert
    await waitFor(() => {
      expect(etfApi.search).toHaveBeenCalledWith({ q: 'TOPIX' });
      expect(result.current.data).toHaveLength(1);
    });
  });

  it('APIエラー時にエラー状態になる', async () => {
    // Arrange
    vi.mocked(etfApi.search).mockRejectedValue(new Error('API Error'));

    // Act
    const { result } = renderHook(() => useETFSearch());
    result.current.search({ q: 'test' });

    // Assert
    await waitFor(() => {
      expect(result.current.error).not.toBeNull();
    });
  });
});
```
